Thank you for installing {{ .Chart.Name }}.

Your application is available at:
{{- if contains "NodePort" .Values.service.type }}
  Get the URL:
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "webapp.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  Get the LoadBalancer IP:
  kubectl get --namespace {{ .Release.Namespace }} svc {{ include "webapp.fullname" . }} -w
{{- else }}
  Run:
  kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ include "webapp.fullname" . }} 8080:{{ .Values.service.port }}
  And visit http://localhost:8080
{{- end }}

To check the status of the pods:
  kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "webapp.name" . }}"
